# Alternative build (the primary build file is Vybrant.pro for qmake6; both produce the same binary).
cmake_minimum_required(VERSION 3.16)
project(Vybrant LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET portaudio-2.0 alsa sndfile samplerate rubberband lame fftw3f jack lilv-0 libcrypto)
find_package(X11 REQUIRED)
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS src/*.cpp src/*.h)
set(CMAKE_AUTORCC ON)
add_executable(vybrant ${SOURCES} resources/app.qrc)
target_include_directories(vybrant PRIVATE src)
target_link_libraries(vybrant PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network PkgConfig::DEPS ${X11_LIBRARIES} ${CMAKE_DL_LIBS} pthread)
target_compile_options(vybrant PRIVATE -Wall -Wno-unused-parameter -Wno-unused-function -fno-strict-aliasing)
target_link_options(vybrant PRIVATE -rdynamic)
set_target_properties(vybrant PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/bin)
