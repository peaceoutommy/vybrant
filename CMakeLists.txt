# Alternative build (the primary build file is Tactum.pro for qmake6; both produce the same binary).
cmake_minimum_required(VERSION 3.16)
project(Tactum LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET portaudio-2.0 alsa sndfile samplerate rubberband lame fftw3f jack lilv-0 libcrypto)
find_package(X11 REQUIRED)
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS src/*.cpp src/*.h)
add_executable(tactum ${SOURCES})
target_include_directories(tactum PRIVATE src)
target_link_libraries(tactum PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network PkgConfig::DEPS ${X11_LIBRARIES} ${CMAKE_DL_LIBS} pthread)
target_compile_options(tactum PRIVATE -Wall -Wno-unused-parameter -Wno-unused-function -fno-strict-aliasing)
target_link_options(tactum PRIVATE -rdynamic)
set_target_properties(tactum PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/bin)
