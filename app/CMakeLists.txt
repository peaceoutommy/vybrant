# Alternative build on Linux (the primary build file is Vybrant.pro for qmake6; both produce the same binary), and the Windows build
cmake_minimum_required(VERSION 3.16)
project(Vybrant LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network)
find_package(PkgConfig REQUIRED)
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS src/*.cpp src/*.h)
if(WIN32)
    # Linux-only for now: LV2 hosting, plugins in their own process, JACK transport; MIDI goes through WinMM instead of ALSA
    list(FILTER SOURCES EXCLUDE REGEX "/src/(plugins/Lv2Plugin\\.(cpp|h)|midi/MidiEngineAlsa\\.cpp)$")
    pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET portaudio-2.0 sndfile samplerate lame libcrypto)
else()
    list(FILTER SOURCES EXCLUDE REGEX "/src/midi/MidiEngineWin\\.cpp$")
    pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET portaudio-2.0 alsa sndfile samplerate lame jack lilv-0 libcrypto)
    find_package(X11 REQUIRED)
endif()
if(WIN32)
    enable_language(RC)
    list(APPEND SOURCES resources/vybrant.rc)   # the exe's own icon
endif()
add_executable(vybrant ${SOURCES} resources/app.qrc)
target_include_directories(vybrant PRIVATE src third_party)
target_compile_options(vybrant PRIVATE -Wall -Wno-unused-parameter -Wno-unused-function -fno-strict-aliasing)
# Cross-compiling from Linux: the host's moc/rcc of the same Qt version (the Windows ones under Wine write paths make can't read)
set(VYBRANT_HOST_QT_BIN "" CACHE PATH "Folder with the build host's moc and rcc when cross-compiling")
if(VYBRANT_HOST_QT_BIN)
    set_target_properties(vybrant PROPERTIES AUTOMOC_EXECUTABLE ${VYBRANT_HOST_QT_BIN}/moc AUTORCC_EXECUTABLE ${VYBRANT_HOST_QT_BIN}/rcc)
endif()
if(WIN32)
    target_link_libraries(vybrant PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network PkgConfig::DEPS winmm)
    set_target_properties(vybrant PROPERTIES WIN32_EXECUTABLE ON OUTPUT_NAME Vybrant)
    # 8 MB stacks, as on Linux (Windows' default is 1-2 MB): Rubber Band 3 overflowed the main thread's while stretching
    target_link_options(vybrant PRIVATE -Wl,--stack,8388608)   # a GUI app: no console window
else()
    target_link_libraries(vybrant PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network PkgConfig::DEPS ${X11_LIBRARIES} ${CMAKE_DL_LIBS} pthread)
    target_link_options(vybrant PRIVATE -rdynamic)
    set_target_properties(vybrant PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/bin)
endif()
